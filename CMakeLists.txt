cmake_minimum_required(VERSION 3.12)
project(MiniGoCpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(minigo_core src/board.cpp src/network.cpp src/search.cpp)
target_include_directories(minigo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(minigo_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(minigo src/main.cpp)
target_link_libraries(minigo PRIVATE minigo_core)
add_executable(minigo_test tests/test.cpp)
target_link_libraries(minigo_test PRIVATE minigo_core)
enable_testing()
add_test(NAME minigo_unit COMMAND minigo_test)
